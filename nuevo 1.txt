function leerDB() {

    );
}

async function cargarTodo() {
    const [r1, r2] = await Promise.all([
        fetch("/api/estudiantes"),
        fetch("/api/notas")
    ]);

    if (!r1.ok || !r2.ok) {
        throw new Error("Error al consultar la API");
    }

    estudiantes = await r1.json();
    notas = await r2.json();

    renderizar();
}

function agregarCelda(fila, valor) {
    const celda = document.createElement("td");
    celda.textContent = valor;
    fila.appendChild(celda);
    return celda;
}

function renderizar() {
    const tablaEstudiantes = $("#listaEstudiantes");
    tablaEstudiantes.replaceChildren();

    estudiantes.forEach(e => {
        const fila = document.createElement("tr");

        [e.matricula, e.nombre, e.correo, e.carrera]
            .forEach(v => agregarCelda(fila, v));

        const celda = agregarCelda(fila, "");
        const boton = document.createElement("button");

        boton.className = "eliminar";
        boton.textContent = "Eliminar";
        boton.type = "button";

        boton.addEventListener("click", () => {
            eliminarEstudiante(e.matricula);
        });

        celda.appendChild(boton);
        tablaEstudiantes.appendChild(fila);
    });

    const selector = $("#estudiante");
    const anterior = selector.value;

    selector.replaceChildren();

    const inicial = document.createElement("option");
    inicial.value = "";
    inicial.textContent = "Seleccione un estudiante";
    selector.appendChild(inicial);

    estudiantes.forEach(e => {
        const opcion = document.createElement("option");
        opcion.value = e.matricula;
        opcion.textContent = e.nombre + " - " + e.matricula;
        selector.appendChild(opcion);
    });

    if (estudiantes.some(e => e.matricula === anterior)) {
        selector.value = anterior;
    }

    const tablaNotas = $("#listaNotas");
    tablaNotas.replaceChildren();

    notas.forEach(n => {
        const fila = document.createElement("tr");

        [n.nombreEstudiante, n.asignatura, n.nota, n.estado]
            .forEach(v => agregarCelda(fila, v));

        const celda = agregarCelda(fila, "");
        const boton = document.createElement("button");

        boton.className = "eliminar";
        boton.textContent = "Eliminar";
        boton.type = "button";

        boton.addEventListener("click", () => eliminarNota(n.id));

        celda.appendChild(boton);
        tablaNotas.appendChild(fila);
    });

    $("#totalEstudiantes").textContent = estudiantes.length;
    $("#totalNotas").textContent = notas.length;

    const promedio = notas.length
        ? notas.reduce((s, n) => s + n.nota, 0) / notas.length
        : 0;

    $("#promedioGeneral").textContent = promedio.toFixed(2);
}

app.post("/api/estudiantes", (req, res) => {
    const { matricula, nombre, correo, carrera } = req.body;

    if (
        ![matricula, nombre, correo, carrera].every(
            v => typeof v === "string" && v.trim()
        )
    ) {
        return res.status(400).json({
            error: "Todos los campos son obligatorios"
        });
    }

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(correo.trim())) {
        return res.status(400).json({
            error: "El correo electrónico no es válido"
        });
    }

    const db = leerDB();

    if (db.estudiantes.some(e => e.matricula === matricula.trim())) {
        return res.status(400).json({
            error: "La matrícula ya está registrada"
        });
    }

    const estudiante = {
        matricula: matricula.trim(),
        nombre: nombre.trim(),
        correo: correo.trim(),
        carrera: carrera.trim()
    };

    db.estudiantes.push(estudiante);
    guardarDB(db);

    res.status(201).json(estudiante);
});

app.delete("/api/estudiantes/:matricula", (req, res) => {
    const db = leerDB();

    const existe = db.estudiantes.some(
        e => e.matricula === req.params.matricula
    );

    if (!existe) {
        return res.status(404).json({
            error: "Estudiante no encontrado"
        });
    }

    db.estudiantes = db.estudiantes.filter(
        e => e.matricula !== req.params.matricula
    );

    db.notas = db.notas.filter(
        n => n.matricula !== req.params.matricula
    );

    guardarDB(db);

    res.status(200).json({
        mensaje: "Estudiante eliminado correctamente"
    });
});

app.get("/api/notas", (req, res) => {
    const db = leerDB();

    const resultado = db.notas.map(n => {
        const estudiante = db.estudiantes.find(
            e => e.matricula === n.matricula
        );

        return {
            ...n,
            nombreEstudiante: estudiante
                ? estudiante.nombre
                : n.matricula,
            estado: n.nota >= 70 ? "Aprobado" : "Reprobado"
        };
    });

    res.status(200).json(resultado);
});

app.post("/api/notas", (req, res) => {
    const { matricula, asignatura, nota } = req.body;

    if (
        typeof matricula !== "string" ||
        !matricula.trim() ||
        typeof asignatura !== "string" ||
        !asignatura.trim() ||
        typeof nota !== "number" ||
        !Number.isFinite(nota)
    ) {
        return res.status(400).json({
            error: "Los campos son obligatorios y la nota debe ser numérica"
        });
    }

    if (nota < 0 || nota > 100) {
        return res.status(400).json({
            error: "La nota debe estar entre 0 y 100"
        });
    }

    const db = leerDB();

    const estudiante = db.estudiantes.find(
        e => e.matricula === matricula
    );

    if (!estudiante) {
        return res.status(404).json({
            error: "El estudiante no existe"
        });
    }

    if (db.notas.some(
        n => n.matricula === matricula &&
             n.asignatura === asignatura.trim()
    )) {
        return res.status(400).json({
            error: "Ya existe una nota para esta asignatura"
        });
    }

    const nuevaNota = {
        id: Date.now().toString() + Math.random().toString(36).slice(2, 7),
        matricula,
        asignatura: asignatura.trim(),
        nota
    };

    db.notas.push(nuevaNota);
    guardarDB(db);

    res.status(201).json(nuevaNota);
});

app.delete("/api/notas/:id", (req, res) => {
    const db = leerDB();

    if (!db.notas.some(n => n.id === req.params.id)) {
        return res.status(404).json({
            error: "Calificación no encontrada"
        });
    }

    db.notas = db.notas.filter(n => n.id !== req.params.id);
    guardarDB(db);

    res.status(200).json({
        mensaje: "Calificación eliminada correctamente"
    });
});

            "http://127.0.0.1:" + PORT + "/api/estudiantes"
        );

        comprobar(
            respuesta.status === 200,
            "API 4 - Consultar estudiantes con HTTP 200"
        );

        await page.close();

    } catch (error) {
        errores++;
        console.error("\\nERROR EN LAS PRUEBAS:", error.message);

    } finally {
        if (browser) {
            await browser.close();
        }

        console.log("\\n================================");
        console.log(" RESULTADO FINAL");
        console.log("================================");

        if (errores === 0) {
            console.log("TODAS LAS PRUEBAS FUERON APROBADAS.");
        } else {
            console.log("PRUEBAS FALLIDAS: " + errores);
        }

        console.log("================================\\n");

        if (errores > 0) {
            process.exitCode = 1;
        }
    }
}
